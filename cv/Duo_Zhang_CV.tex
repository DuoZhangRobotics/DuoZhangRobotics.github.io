\section*{Research Experience}
\entry{Rutgers University, Algorithmic Robotics \& Control Lab}{September 2023--Present}
Graduate Assistant \enspace $\vert$ \enspace Advisor: Prof. Jingjin Yu

\begin{itemize}
  \project{Geometric and online motion planning (RVG / dRVG).}{Developed RVG's
  rotation-stacked visibility graphs for resolution-complete, asymptotically optimal
  multi-query planning of polygonal robots. Extended its geometric roadmaps to
  unknown static environments with dRVG's incremental graph merging and
  quadtree-guided exploration; validated online navigation on microMVP.
  \paperref{rvg}, \paperref{drvg}}

  \project{GPU-accelerated manipulation and space-time planning.}{Integrated
  dependency-graph task planning with GPU-accelerated synchronous motion generation
  in \textbf{SDAR}: \textbf{100\% end-to-end success} on evaluated non-monotone
  rearrangement tasks in simulation; validated on dual UR5e arms. \paperref{sdar}
  Developed \textbf{ST-pRRTC}, a GPU-parallel space-time RRT-Connect planner with
  a shared forward tree and adaptive goal-time forests for known moving-obstacle
  trajectories; demonstrated UR5e motion among moving Crazyflie quadrotors.
  \paperref{stprrtc}}

  \project{Free-floating space manipulation (FAVOR).}{Developed B\'ezier
  joint-velocity optimization with analytical recursive sensitivities for reaching,
  tracking, and prescribed-time interception under coupled spacecraft--arm dynamics.
  Achieved \textbf{99.8\% point-to-point success} at \textbf{1.78\,s mean computation}
  in simulation with a seven-DoF arm and five spacecraft models; timing includes
  failures and timeouts.
  \paperref{favor}}
\end{itemize}
